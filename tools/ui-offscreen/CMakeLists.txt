cmake_minimum_required(VERSION 3.16)
project(ReplayNESUIOffscreen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(REPLAYNES_BUILD_TESTS ON CACHE BOOL "" FORCE)
get_filename_component(REPLAYNES_ROOT ../.. ABSOLUTE)
add_subdirectory(${REPLAYNES_ROOT} replaynes)
find_package(SDL3 CONFIG REQUIRED)
set(RNL_SDL3_TARGET SDL3::SDL3-static)
set(RNL_APP_VERSION "offscreen-test")
set(RNL_APP_FEATURES RNL_HAVE_STEAM=0 RNL_HAVE_MP4_EXPORT=0 RNL_HAVE_CRT=0)
include(${REPLAYNES_ROOT}/apps/desktop/app.cmake)
add_executable(replaynes-ui-offscreen main.cpp ${REPLAYNES_ROOT}/apps/desktop/src/export/export_job.cpp ${REPLAYNES_ROOT}/apps/desktop/src/export/export_frame.cpp)
target_link_libraries(replaynes-ui-offscreen PRIVATE rnl_app)
