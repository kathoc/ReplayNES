cmake_minimum_required(VERSION 3.16)
project(ReplayNES VERSION 0.5.2 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(REPLAYNES_BUILD_TESTS "Build engine tests" ON)
option(REPLAYNES_BUILD_CLI "Build replaynes-cli" ON)
option(REPLAYNES_BUILD_FRONTEND "Build the shared frontend core (replaynes_frontend)" ON)
option(REPLAYNES_BUILD_WINDOWS_PROBE "Build replaynes-winprobe (Windows graphics/audio/input API report)" ${WIN32})
option(REPLAYNES_BUILD_DESKTOP "Build the shared desktop frontend logic (apps/desktop) and its tests" ON)
option(REPLAYNES_BUILD_WINDOWS_APP "Windows: build the frontend replaynes-win (SDL3 from third_party/SDL + Direct3D 11)" ${WIN32})

include(cmake/NestopiaCore.cmake)
add_subdirectory(engine)
if(REPLAYNES_BUILD_FRONTEND)
  add_subdirectory(frontend)
endif()
if(REPLAYNES_BUILD_CLI)
  add_subdirectory(tools/replaynes-cli)
endif()
if(REPLAYNES_BUILD_WINDOWS_PROBE AND WIN32)
  add_subdirectory(tools/windows-probe)
endif()
if(REPLAYNES_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
# Shared desktop frontend (Linux / Windows): logic library + tests on every platform; the Windows
# app on Windows (the Linux app is the standalone project apps/linux, which adds this tree).
if(REPLAYNES_BUILD_FRONTEND AND REPLAYNES_BUILD_DESKTOP)
  enable_testing()  # test_desktop_frontend (also when this tree is added by apps/linux)
  add_subdirectory(apps/desktop)
  if(WIN32 AND REPLAYNES_BUILD_WINDOWS_APP)
    add_subdirectory(apps/windows)
  endif()
endif()
